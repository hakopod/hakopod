func freePolicyFixture(t *testing.T) (metav1.ObjectMeta, corev1.PodSpec) {
	t.Helper()
	root := os.Getenv("HAKOPOD_ORACLE_FREE_CANDIDATE_DIR")
	if root == "" {
		t.Fatal("Oracle Free policy regression requires the exported Hakopod candidate")
	}
	raw, err := os.ReadFile(filepath.Join(root, "database.json"))
	if err != nil {
		t.Fatal(err)
	}
	var object struct {
		Metadata metav1.ObjectMeta `json:"metadata"`
	}
	if err = json.Unmarshal(raw, &object); err != nil {
		t.Fatal(err)
	}
	var policy struct {
		PodSpec corev1.PodSpec `json:"podSpec"`
	}
	if err = json.Unmarshal([]byte(object.Metadata.Annotations["hakopod.io/oracle-free-pod-policy"]), &policy); err != nil {
		t.Fatal(err)
	}
	return object.Metadata, policy.PodSpec
}

func TestHakopodOracleFreePolicyRejectsBoundaryChanges(t *testing.T) {
	owner, expected := freePolicyFixture(t)
	newPod := func() *corev1.Pod {
		return &corev1.Pod{Spec: corev1.PodSpec{Containers: []corev1.Container{{Name: "database", Image: hakopodFreeImage}}}}
	}
	valid := newPod()
	if err := ApplyHakopodFreePodPolicy(valid, owner); err != nil {
		t.Fatal(err)
	}
	if len(valid.Spec.InitContainers) != 0 || valid.Spec.Containers[0].Image != hakopodFreeImage || valid.Labels["hakopod.io/oracle-member"] != "true" || valid.Annotations["hakopod.io/oracle-identity"] != owner.Annotations["hakopod.io/oracle-identity"] {
		t.Fatal("Free bootstrap or identity propagation changed")
	}
	for name, change := range map[string]func(*corev1.PodSpec){
		"host networking":     func(p *corev1.PodSpec) { p.HostNetwork = true },
		"operator identity":   func(p *corev1.PodSpec) { p.ServiceAccountName = "database-oracle-free-operator" },
		"token":               func(p *corev1.PodSpec) { yes := true; p.AutomountServiceAccountToken = &yes },
		"mutable image":       func(p *corev1.PodSpec) { p.Containers[0].Image = "oracle:latest" },
		"secret environment":  func(p *corev1.PodSpec) { p.Containers[0].Env = []corev1.EnvVar{{Name: "PASSWORD", Value: "private"}} },
		"extra command":       func(p *corev1.PodSpec) { p.Containers[0].Args = []string{"unexpected"} },
		"foreign backup":      func(p *corev1.PodSpec) { p.Volumes[len(p.Volumes)-1].PersistentVolumeClaim.ClaimName = "foreign" },
		"foreign secret":      func(p *corev1.PodSpec) { p.Volumes[0].Secret.SecretName = "foreign" },
		"supplemental groups": func(p *corev1.PodSpec) { p.SecurityContext.SupplementalGroups = []int64{0} },
		"pod SELinux": func(p *corev1.PodSpec) {
			p.SecurityContext.SELinuxOptions = &corev1.SELinuxOptions{Type: "unconfined_t"}
		},
		"unmasked proc": func(p *corev1.PodSpec) {
			value := corev1.UnmaskedProcMount
			p.Containers[0].SecurityContext.ProcMount = &value
		},
		"container seccomp": func(p *corev1.PodSpec) {
			p.Containers[0].SecurityContext.SeccompProfile = &corev1.SeccompProfile{Type: corev1.SeccompProfileTypeUnconfined}
		},
		"secret item paths": func(p *corev1.PodSpec) {
			p.Volumes[0].Secret.Items = []corev1.KeyToPath{{Key: "tls.crt", Path: "different"}}
		},
		"optional secret": func(p *corev1.PodSpec) { p.Volumes[0].Secret.Optional = freePtr(true) },
		"config item paths": func(p *corev1.PodSpec) {
			for i := range p.Volumes {
				if p.Volumes[i].ConfigMap != nil {
					p.Volumes[i].ConfigMap.Items = []corev1.KeyToPath{{Key: "start.sh", Path: "different"}}
					break
				}
			}
		},
		"optional config": func(p *corev1.PodSpec) {
			for i := range p.Volumes {
				if p.Volumes[i].ConfigMap != nil {
					p.Volumes[i].ConfigMap.Optional = freePtr(true)
					break
				}
			}
		},
		"temporary memory": func(p *corev1.PodSpec) {
			for i := range p.Volumes {
				if p.Volumes[i].Name == "temporary" {
					p.Volumes[i].EmptyDir.Medium = corev1.StorageMediumMemory
					break
				}
			}
		},
		"oversized temporary": func(p *corev1.PodSpec) {
			for i := range p.Volumes {
				if p.Volumes[i].Name == "temporary" {
					p.Volumes[i].EmptyDir.SizeLimit = freePtr(resource.MustParse("1Ti"))
					break
				}
			}
		},
		"additional volume source": func(p *corev1.PodSpec) { p.Volumes[0].HostPath = &corev1.HostPathVolumeSource{Path: "/"} },
		"unbounded temporary": func(p *corev1.PodSpec) {
			for i := range p.Volumes {
				if p.Volumes[i].EmptyDir != nil {
					p.Volumes[i].EmptyDir.SizeLimit = nil
					break
				}
			}
		},
	} {
		t.Run(name, func(t *testing.T) {
			changed := expected.DeepCopy()
			change(changed)
			metadata := owner.DeepCopy()
			raw, _ := json.Marshal(map[string]any{"profile": HakopodOracleFreeProfile, "podSpec": changed})
			metadata.Annotations["hakopod.io/oracle-free-pod-policy"] = string(raw)
			if ApplyHakopodFreePodPolicy(newPod(), *metadata) == nil {
				t.Fatal("accepted changed workload boundary")
			}
		})
	}
	for _, raw := range []string{"{}", strings.Repeat("x", 33<<10), owner.Annotations["hakopod.io/oracle-free-pod-policy"] + "{}"} {
		metadata := owner.DeepCopy()
		metadata.Annotations["hakopod.io/oracle-free-pod-policy"] = raw
		if ApplyHakopodFreePodPolicy(newPod(), *metadata) == nil {
			t.Fatal("accepted malformed pod policy")
		}
	}
	foreign := owner.DeepCopy()
	foreign.Namespace = "another-namespace"
	if ApplyHakopodFreePodPolicy(newPod(), *foreign) == nil {
		t.Fatal("accepted foreign namespace")
	}
}

func TestHakopodOracleFreeStorageNeverAdoptsOrReplacesClaims(t *testing.T) {
	owner, _ := freePolicyFixture(t)
	owner.UID = "root-uid"
	yes := true
	class := "local-path"
	claim := &corev1.PersistentVolumeClaim{ObjectMeta: metav1.ObjectMeta{Name: "database", Namespace: owner.Namespace, Finalizers: []string{HakopodOracleFreeStorageFinalizer}, OwnerReferences: []metav1.OwnerReference{{APIVersion: "database.oracle.com/v4", Kind: "SingleInstanceDatabase", Name: owner.Name, UID: owner.UID, Controller: &yes, BlockOwnerDeletion: &yes}}}, Spec: corev1.PersistentVolumeClaimSpec{StorageClassName: &class, AccessModes: []corev1.PersistentVolumeAccessMode{corev1.ReadWriteOnce}, Resources: corev1.VolumeResourceRequirements{Requests: corev1.ResourceList{corev1.ResourceStorage: resource.MustParse("10Gi")}}}}
	if err := ValidateHakopodFreePVC(claim, owner, "database", class, "10Gi"); err != nil {
		t.Fatal(err)
	}
	for name, change := range map[string]func(*corev1.PersistentVolumeClaim){
		"foreign owner":        func(p *corev1.PersistentVolumeClaim) { p.OwnerReferences[0].UID = "foreign" },
		"no owner":             func(p *corev1.PersistentVolumeClaim) { p.OwnerReferences = nil },
		"no storage finalizer": func(p *corev1.PersistentVolumeClaim) { p.Finalizers = nil },
		"nonblocking owner":    func(p *corev1.PersistentVolumeClaim) { p.OwnerReferences[0].BlockOwnerDeletion = nil },
		"empty modes":          func(p *corev1.PersistentVolumeClaim) { p.Spec.AccessModes = nil },
		"size drift": func(p *corev1.PersistentVolumeClaim) {
			p.Spec.Resources.Requests[corev1.ResourceStorage] = resource.MustParse("11Gi")
		},
		"class drift": func(p *corev1.PersistentVolumeClaim) { value := "different"; p.Spec.StorageClassName = &value },
		"clone source": func(p *corev1.PersistentVolumeClaim) {
			p.Spec.DataSource = &corev1.TypedLocalObjectReference{Kind: "PersistentVolumeClaim", Name: "foreign"}
		},
	} {
		t.Run(name, func(t *testing.T) {
			changed := claim.DeepCopy()
			change(changed)
			if ValidateHakopodFreePVC(changed, owner, "database", class, "10Gi") == nil {
				t.Fatal("accepted changed data claim")
			}
		})
	}
}

func TestHakopodOracleFreeObservedMemberRequiresCurrentOwner(t *testing.T) {
	owner, _ := freePolicyFixture(t)
	owner.UID = "root-uid"
	pod := &corev1.Pod{Spec: corev1.PodSpec{Containers: []corev1.Container{{Image: hakopodFreeImage}}}}
	if err := ApplyHakopodFreePodPolicy(pod, owner); err != nil {
		t.Fatal(err)
	}
	pod.Name = "database-member"
	pod.Namespace = owner.Namespace
	pod.UID = "pod-uid"
	yes := true
	pod.OwnerReferences = []metav1.OwnerReference{{APIVersion: "database.oracle.com/v4", Kind: "SingleInstanceDatabase", Name: owner.Name, UID: owner.UID, Controller: &yes}}
	if err := ValidateHakopodFreeObservedMember(pod, owner); err != nil {
		t.Fatal(err)
	}
	for name, change := range map[string]func(*corev1.Pod){
		"foreign owner":       func(p *corev1.Pod) { p.OwnerReferences[0].UID = "foreign" },
		"no owner":            func(p *corev1.Pod) { p.OwnerReferences = nil },
		"wrong image":         func(p *corev1.Pod) { p.Spec.Containers[0].Image = "foreign:image" },
		"extra exec argument": func(p *corev1.Pod) { p.Spec.Containers[0].Args = []string{"extra"} },
		"token":               func(p *corev1.Pod) { p.Spec.AutomountServiceAccountToken = &yes },
		"old identity":        func(p *corev1.Pod) { p.Annotations["hakopod.io/oracle-identity"] = "old" },
	} {
		t.Run(name, func(t *testing.T) {
			changed := pod.DeepCopy()
			change(changed)
			if ValidateHakopodFreeObservedMember(changed, owner) == nil {
				t.Fatal("operator accepted an unsafe exec target")
			}
		})
	}
}
